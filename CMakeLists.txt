cmake_minimum_required(VERSION 3.5)

project(PosterChef VERSION 0.1 LANGUAGES C CXX)

# include(ExternalProject)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(PkgConfig REQUIRED)
pkg_check_modules(libimobiledevice REQUIRED IMPORTED_TARGET libimobiledevice-1.0)
pkg_check_modules(libimobiledevice-glue REQUIRED IMPORTED_TARGET libimobiledevice-glue-1.0)
pkg_check_modules(openssl REQUIRED IMPORTED_TARGET openssl)

add_library(libimobiledevice STATIC IMPORTED)
set_target_properties(libimobiledevice PROPERTIES
    IMPORTED_LOCATION ${CMAKE_BINARY_DIR}/vendor/libimobiledevice/
)

add_library(openssl STATIC IMPORTED)
set_target_properties(openssl PROPERTIES
    IMPORTED_LOCATION ${CMAKE_BINARY_DIR}/vendor/openssl/
)

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets LinguistTools)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets LinguistTools)

set(TS_FILES PosterChef_en_US.ts)

set(PROJECT_SOURCES
        main.cpp
        mainwindow.cpp
        mainwindow.h
        mainwindow.ui
        ${TS_FILES}
)

set(app_icon_resource_windows "${CMAKE_CURRENT_SOURCE_DIR}/credits/PosterChef.ico")
# The MACOSX_BUNDLE_ICON_FILE variable is added to the Info.plist
# generated by CMake. This variable contains the .icns file name,
# without the path.
set(MACOSX_BUNDLE_ICON_FILE PosterChef.icns)

# And the following tells CMake where to find and install the file itself.
set(app_icon_macos "${CMAKE_CURRENT_SOURCE_DIR}/credits/PosterChef.icns")
set_source_files_properties(${app_icon_macos} PROPERTIES
       MACOSX_PACKAGE_LOCATION "Resources")

if(${QT_VERSION_MAJOR} GREATER_EQUAL 6)
    qt_add_executable(PosterChef
        MANUAL_FINALIZATION
        ${PROJECT_SOURCES}

        resources.qrc
        devicemanager.h devicemanager.cpp
        utils.h
        credits/LeminLimez.png
        posterboardmanager.h posterboardmanager.cpp
        tendiesfile.h tendiesfile.cpp
        CreateBackup.cpp CreateBackup.h
        proc_modules/idevicebackup2.c proc_modules/idevicebackup2.h proc_modules/endianness.h
        ${app_icon_resource_windows}
        ${app_icon_macos}
    )
# Define target properties for Android with Qt 6 as:
#    set_property(TARGET PosterChef APPEND PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR
#                 ${CMAKE_CURRENT_SOURCE_DIR}/android)
# For more information, see https://doc.qt.io/qt-6/qt-add-executable.html#target-creation

    qt_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
else()
    if(ANDROID)
        add_library(PosterChef SHARED
            ${PROJECT_SOURCES}
        )
# Define properties for Android with Qt 5 after find_package() calls as:
#    set(ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    else()
        add_executable(PosterChef
            ${PROJECT_SOURCES}
        )
    endif()

    qt5_create_translation(QM_FILES ${CMAKE_SOURCE_DIR} ${TS_FILES})
endif()

# target_link_libraries(idevicebackup2 PRIVATE PkgConfig::libimobiledevice)
target_link_libraries(PosterChef PRIVATE Qt${QT_VERSION_MAJOR}::Widgets)
target_link_libraries(PosterChef PRIVATE PkgConfig::libimobiledevice)
target_link_libraries(PosterChef PRIVATE PkgConfig::libimobiledevice-glue)
target_link_libraries(PosterChef PRIVATE PkgConfig::openssl)

# Qt for iOS sets MACOSX_BUNDLE_GUI_IDENTIFIER automatically since Qt 6.1.
# If you are developing for iOS or macOS you should consider setting an
# explicit, fixed bundle identifier manually though.
if(${QT_VERSION} VERSION_LESS 6.1.0)
  set(BUNDLE_ID_OPTION MACOSX_BUNDLE_GUI_IDENTIFIER com.leemin.PosterChef)
endif()
set_target_properties(PosterChef PROPERTIES
    ${BUNDLE_ID_OPTION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
)

include(GNUInstallDirs)
install(TARGETS PosterChef
    BUNDLE DESTINATION .
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

if(QT_VERSION_MAJOR EQUAL 6)
    qt_finalize_executable(PosterChef)
endif()
